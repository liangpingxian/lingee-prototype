import { Component, createEffect, createSignal, For, Match, onCleanup, Show, Switch } from "solid-js"
import { FileIcon } from "@kcode/ui/file-icon"
import { Icon } from "@kcode/ui/icon"
import { Spinner } from "@kcode/ui/spinner"
import { Tooltip } from "@kcode/ui/tooltip"
import { getDirectory, getFilename } from "@opencode-ai/core/util/path"
import { shouldShowSkillsLoading, shouldShowSlashList } from "./slash-trigger"
import "./slash-popover.css"

export type AtOption =
  | { type: "agent"; name: string; display: string; id?: string }
  | { type: "file"; path: string; display: string; recent?: boolean }
  | { type: "rule"; path: string; name: string; display: string; ruleType: string; scope: "global" | "project" }

export interface SlashCommand {
  id: string
  trigger: string
  title: string
  description?: string
  keybind?: string
  type: "builtin" | "custom"
  source?: "alias" | "command" | "mcp" | "skill"
  /** For source === "skill": where the skill originated. */
  skillScope?: "builtin" | "user" | "project"
}

export type SlashGroup = { category: string; items: SlashCommand[] }

type PromptPopoverProps = {
  popover: "at" | "slash" | null
  setSlashPopoverRef: (el: HTMLDivElement) => void
  isNewSession?: boolean
  newSessionPopoverWidth?: number
  atFlat: AtOption[]
  atActive?: string
  atKey: (item: AtOption) => string
  setAtActive: (id: string) => void
  onAtSelect: (item: AtOption) => void
  slashFlat: SlashCommand[]
  slashGrouped: SlashGroup[]
  slashActive?: string
  setSlashActive: (id: string) => void
  onSlashSelect: (item: SlashCommand) => void
  commandKeybind: (id: string) => string | undefined
  skillsLoading?: boolean
  t: (key: string) => string
}

const SKILL_GROUP = "skills"

// @ 菜单分组：顺序与 sortGroupsBy 一致（agent → recent → rule → file），用于粘性组头。
const atGroup = (item: AtOption) => {
  if (item.type === "agent") return "agent"
  if (item.type === "rule") return "rule"
  if (item.recent) return "recent"
  return "file"
}

export const PromptPopover: Component<PromptPopoverProps> = (props) => {
  const newSessionStyle = () =>
    props.isNewSession && props.newSessionPopoverWidth
      ? {
          width: `${props.newSessionPopoverWidth}px`,
          "max-width": `${props.newSessionPopoverWidth}px`,
        }
      : undefined

  // 动态测量浮窗可用高度：由于 -translate-y-full，面板 bottom 边缘始终固定在
  // 触发点位置（父元素顶部 - 8px），与面板自身高度无关。据此计算从 bottom 到
  // 视口顶部的可用空间，减去 20px 最小间距，作为 max-height 上限。
  const [dynamicMaxHeight, setDynamicMaxHeight] = createSignal<string | undefined>(undefined)
  let popoverEl: HTMLDivElement | undefined

  const measureMaxHeight = () => {
    if (!popoverEl) return
    const rect = popoverEl.getBoundingClientRect()
    const available = rect.bottom - 20
    const originalMax = props.isNewSession ? 256 : 320
    const maxH = Math.min(available, originalMax)
    setDynamicMaxHeight(maxH > 0 ? `${Math.floor(maxH)}px` : undefined)
  }

  createEffect(() => {
    if (!props.popover) {
      setDynamicMaxHeight(undefined)
      return
    }
    requestAnimationFrame(measureMaxHeight)
    const handler = () => requestAnimationFrame(measureMaxHeight)
    window.addEventListener("resize", handler)
    onCleanup(() => window.removeEventListener("resize", handler))
  })

  // Show loading placeholder only when skills are still loading AND no skill
  // items have arrived yet — prevents duplicate headers during the overlap.
  // 保持为函数（而非提前求值的常量）：每次调用都读取 props，SolidJS 才能建立
  // 响应式依赖；提前求值会让弹层卡在首次渲染的状态不再更新。
  const showSkillsLoading = () =>
    shouldShowSkillsLoading({ skillsLoading: !!props.skillsLoading, groups: props.slashGrouped })

  const badgeLabel = (cmd: SlashCommand) => {
    if (cmd.skillScope === "builtin") return props.t("prompt.slash.badge.builtin")
    if (cmd.skillScope === "user") return props.t("prompt.slash.badge.user")
    if (cmd.skillScope === "project") return props.t("prompt.slash.badge.project")
    if (cmd.source === "mcp") return props.t("prompt.slash.badge.mcp")
    // source === "skill" 但 skillScope 未由后端填充（如后端未重启）时，兜底为"内置"
    if (cmd.source === "skill" || cmd.source === "alias") return props.t("prompt.slash.badge.builtin")
    return props.t("prompt.slash.badge.custom")
  }

  const showBadge = (cmd: SlashCommand) =>
    cmd.type === "custom" && cmd.source !== "command"

  return (
    // slash 会话有效时始终保留弹层外壳。候选加载/过滤的中间空值
    // 不应让整个弹层消失；真正无匹配时由内层 fallback 给出明确空态。
    <Show when={!!props.popover}>
      <div
        ref={(el) => {
          popoverEl = el
          if (props.popover === "slash") props.setSlashPopoverRef(el)
        }}
        class="absolute left-0 -top-2 -translate-y-full origin-bottom-left min-h-10 z-[200]
                 overflow-auto no-scrollbar flex flex-col rounded-[20px] p-1
                 bg-surface-raised-stronger-non-alpha shadow-[var(--shadow-lg-border-base)]"
        classList={{
          "inset-x-0 max-h-80": !props.isNewSession,
          "max-h-64": !!props.isNewSession,
        }}
        style={{ ...(newSessionStyle() ?? {}), ...(dynamicMaxHeight() ? { "max-height": dynamicMaxHeight()! } : {}) }}
        onMouseDown={(e) => e.preventDefault()}
      >
        <Switch>
          <Match when={props.popover === "at"}>
            <div class="flex flex-col gap-0.5">
              <Show
                when={props.atFlat.length > 0}
                fallback={<div class="text-text-weak px-2 py-1">{props.t("prompt.popover.emptyResults")}</div>}
              >
                <For each={props.atFlat.slice(0, 10)}>
                  {(item, index) => {
                    const key = props.atKey(item)
                    const group = atGroup(item)
                    const prev = index() > 0 ? props.atFlat[index() - 1] : undefined
                    const header =
                      !prev || atGroup(prev) !== group ? (
                        <div class="sticky -top-1 z-10 px-2 py-1 bg-surface-raised-stronger-non-alpha">
                          <span class="text-12-medium text-text-weak">{props.t(`prompt.at.group.${group}`)}</span>
                        </div>
                      ) : undefined

                    if (item.type === "agent") {
                      return (
                        <>
                          {header}
                          <button
                            class="w-full flex items-center gap-x-2 rounded-full h-8 px-2"
                            classList={{ "bg-surface-raised-base-hover": props.atActive === key }}
                            onClick={() => props.onAtSelect(item)}
                            onMouseEnter={() => props.setAtActive(key)}
                          >
                            <Icon name="brain" size="small" class="text-icon-info-active shrink-0" />
                            <span class="text-14-regular text-text-strong whitespace-nowrap">@{item.name}</span>
                          </button>
                        </>
                      )
                    }

                    if (item.type === "rule") {
                      return (
                        <>
                          {header}
                          <button
                            class="w-full flex items-center gap-x-2 rounded-full h-8 px-2"
                            classList={{ "bg-surface-raised-base-hover": props.atActive === key }}
                            onClick={() => props.onAtSelect(item)}
                            onMouseEnter={() => props.setAtActive(key)}
                          >
                            <Icon name="book-open" size="small" class="text-icon-info-active shrink-0" />
                            <span class="text-14-regular text-text-strong truncate" title={item.path}>
                              {item.name}
                            </span>
                            <span class="kcode-slash-badge kcode-slash-badge--builtin shrink-0">
                              {props.t(`prompt.at.rule.type.${item.ruleType}`)}
                            </span>
                          </button>
                        </>
                      )
                    }

                    const isDirectory = item.path.endsWith("/")
                    const directory = isDirectory ? item.path : getDirectory(item.path)
                    const filename = isDirectory ? "" : getFilename(item.path)

                    return (
                      <>
                        {header}
                        <button
                          class="w-full flex items-center gap-x-2 rounded-full h-8 px-2"
                          classList={{ "bg-surface-raised-base-hover": props.atActive === key }}
                          onClick={() => props.onAtSelect(item)}
                          onMouseEnter={() => props.setAtActive(key)}
                        >
                          <FileIcon node={{ path: item.path, type: "file" }} class="shrink-0 size-4" />
                          <div class="flex items-center text-14-regular min-w-0">
                            <span class="text-text-weak whitespace-nowrap truncate min-w-0">{directory}</span>
                            <Show when={!isDirectory}>
                              <span class="text-text-strong whitespace-nowrap">{filename}</span>
                            </Show>
                          </div>
                        </button>
                      </>
                    )
                  }}
                </For>
              </Show>
            </div>
          </Match>
          <Match when={props.popover === "slash"}>
            {/* 外层必须同时放行技能加载态，否则候选被收窄为「仅技能」时，启动阶段
                技能尚未返回会让 slashFlat 为空而错误显示「无指令」。
                判定写在 when 表达式内以保留 SolidJS 响应式。 */}
            <Show
              when={shouldShowSlashList({
                flatCount: props.slashFlat.length,
                skillsLoading: !!props.skillsLoading,
                groups: props.slashGrouped,
              })}
              fallback={<div class="flex items-center text-text-weak px-2 min-h-10">{props.t("prompt.popover.emptyCommands")}</div>}
            >
              <Show when={showSkillsLoading()}>
                <div class="flex flex-col">
                  <div
                    class="sticky -top-1 z-10 flex items-center gap-1.5 px-2 py-1
                           bg-surface-raised-stronger-non-alpha"
                  >
                    <span class="text-12-medium text-text-weak">
                      {props.t("prompt.slash.group.skills")}
                    </span>
                    <Tooltip
                      placement="top"
                      contentClass="max-w-[280px] text-13-regular"
                      value={props.t("prompt.slash.skills.tooltip")}
                    >
                      <Icon name="info" size="small" class="text-icon-weak cursor-help" />
                    </Tooltip>
                  </div>
                  <div class="flex items-center gap-2 px-2 py-1.5 text-text-weak">
                    <Spinner class="size-3.5 shrink-0" />
                    <span class="text-14-regular">{props.t("prompt.slash.skills.loading")}</span>
                  </div>
                </div>
              </Show>
              <For each={props.slashGrouped}>
                {(group) => (
                  <div class="flex flex-col gap-0.5">
                    {/* Sticky group header — -top-1 offsets the container's 4px
                        padding so the header pins flush to the visible scroll edge. */}
                    <div
                      class="sticky -top-1 z-10 flex items-center gap-1.5 px-2 py-1
                             bg-surface-raised-stronger-non-alpha"
                    >
                      <span class="text-12-medium text-text-weak">
                        {group.category === SKILL_GROUP
                          ? props.t("prompt.slash.group.skills")
                          : props.t("prompt.slash.group.commands")}
                      </span>
                      <Show when={group.category === SKILL_GROUP}>
                        <Tooltip
                          placement="top"
                          contentClass="max-w-[280px] text-13-regular"
                          value={props.t("prompt.slash.skills.tooltip")}
                        >
                          <Icon name="info" size="small" class="text-icon-weak cursor-help" />
                        </Tooltip>
                      </Show>
                    </div>
                    <For each={group.items}>
                      {(cmd) => (
                        <button
                          data-slash-id={cmd.id}
                          classList={{
                            "w-full flex items-center justify-between gap-4 rounded-full h-8 px-2": true,
                            "bg-surface-raised-base-hover": props.slashActive === cmd.id,
                          }}
                          onClick={() => props.onSlashSelect(cmd)}
                          onMouseEnter={() => props.setSlashActive(cmd.id)}
                        >
                          <div class="flex items-center gap-2 min-w-0">
                            <span class="text-14-regular text-text-strong whitespace-nowrap">{cmd.title}</span>
                            <Show when={cmd.description}>
                              <span class="text-14-regular text-text-weak truncate">{cmd.description}</span>
                            </Show>
                          </div>
                          <div class="flex items-center gap-2 shrink-0">
                            <Show when={showBadge(cmd)}>
                              <span
                                class="kcode-slash-badge"
                                classList={{
                                  "kcode-slash-badge--project": cmd.skillScope === "project",
                                  "kcode-slash-badge--user": cmd.skillScope === "user",
                                  "kcode-slash-badge--builtin": cmd.skillScope !== "project" && cmd.skillScope !== "user",
                                }}
                              >
                                {badgeLabel(cmd)}
                              </span>
                            </Show>
                            <Show when={props.commandKeybind(cmd.id)}>
                              <span class="text-12-regular text-text-weak">{props.commandKeybind(cmd.id)}</span>
                            </Show>
                          </div>
                        </button>
                      )}
                    </For>
                  </div>
                )}
              </For>
            </Show>
          </Match>
        </Switch>
      </div>
    </Show>
  )
}
