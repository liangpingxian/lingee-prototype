/** Extract the skill-dev name from an agent part.
 *  Slash-command pills carry name=trigger; @-mention pills carry id=skill-name
 *  with name=display-label. Checks both fields against SKILL_DEV_MODE_MAP. */
export const getSkillDevName = (part: { type: string; name?: string; id?: string }): string | undefined => {
  if (part.type !== "agent") return undefined
  if (part.name && part.name in SKILL_DEV_MODE_MAP) return part.name
  if (part.id && part.id in SKILL_DEV_MODE_MAP) return part.id
  return undefined
}

        title: cmd.source === "skill" ? (SKILL_DISPLAY_NAME_MAP[cmd.name] ?? cmd.name) : cmd.name,
        description: cmd.description,
        type: "custom" as const,
        source: cmd.source,
        skillScope: cmd.skillScope,

    // Only show it once the backend command list has loaded (i.e. real skill
    // commands are present) AND the real standalone-build skill is not already
    // in the list.  Without the first guard the alias — which carries
    // skillScope: "builtin" — is the *only* "built-in" entry visible during the
    // 1–2s bootstrap window before command.list() resolves, giving the false
    // impression that just one built-in skill is available.  Without the second
    // guard the alias duplicates the real skill entry.
    //
    // Guards MUST be computed from backendCmds (before adding the alias):
    // hasStandaloneBuild checks trigger === "standalone-build", and the alias
    // itself carries that trigger — computing from the merged array would make
    // the guard always true and the alias would never show.
    const hasSkillCommands = backendCmds.some((cmd) => cmd.source === "skill")
    const hasStandaloneBuild = backendCmds.some((cmd) => cmd.trigger === "standalone-build")

    // Merge alias into the array BEFORE sorting so it participates in the
    // skillScope-based ordering instead of being tacked on at the end.
    // The sort treats source === "alias" as skill-like (same tier as "skill")
    // so the alias lands among its peers by skillScope, not after non-skill
    // commands like MCP prompts.
    const isSkillLike = (s: string | undefined) => s === "skill" || s === "alias"

    const custom = [
      ...backendCmds,
      ...(hasSkillCommands && !hasStandaloneBuild && canShowSkillTab(licensePlan(), "standalone-build")
        ? [{
            id: "alias.standalone-build",
            trigger: "standalone-build",
            title: "expert-team-build",
            description: "Legacy alias for expert-team-build",
            type: "custom" as const,
            source: "alias" as const,
            skillScope: "builtin" as const,
          }]
        : []),

      // 技能（含 alias）排在最前，其余按后端返回顺序保持不变
      if (isSkillLike(a.source) && !isSkillLike(b.source)) return -1
      if (!isSkillLike(a.source) && isSkillLike(b.source)) return 1
      // 技能内部按 skillScope 排序：内置 → 用户 → 项目

        const rank = (s: string | undefined) =>
          s === "builtin" ? 0 : s === "user" ? 1 : s === "project" ? 2 : 3
        return rank(a.skillScope) - rank(b.skillScope)
      }
      return 0
    })

    return [
      ...custom,
      ...builtin,
    ]

    closePopover()
  }

        const text = node.textContent ?? ""
        if (!text.includes("\u200B")) return true
        if (text !== "\u200B") return false

        const prev = node.previousSibling
        const next = node.nextSibling
        const prevIsBr = prev?.nodeType === Node.ELEMENT_NODE && (prev as HTMLElement).tagName === "BR"
        return !!prevIsBr && !next
      }
      if (node.nodeType !== Node.ELEMENT_NODE) return false
      const el = node as HTMLElement
      if (el.dataset.type === "file") return true
      if (el.dataset.type === "agent") return true
      if (el.dataset.type === "rule") return true
      return el.tagName === "BR"

    // pill/BR and no trailing \u200B is NOT in the rendered state — without it,
    // some webviews (e.g. lingee-yzj) cannot place the caret at all, causing the
    // editor to lose focus and become unfocusable.
    const last = children[children.length - 1]
    if (last?.nodeType === Node.ELEMENT_NODE) {
      const el = last as HTMLElement
      if (el.tagName === "BR" || el.dataset.type === "file" || el.dataset.type === "agent" || el.dataset.type === "rule")
        return false
    }

    return true

    // 2) 必须在这个分支里而不是 clearEditor 里 —— 用户输入触发的 DOM 规范化
    //    （上面 mirror.input 分支）同样经由 clearEditor，在那里清会留下

    resetSlashSession()

    // External state change (session switch, revert, edit, …) — restore from
    // the persisted cursor instead of the (now stale) DOM selection.
    renderEditorWithCursor(input)
  }

  const parseFromDOM = (): Prompt => {
    const parts: Prompt = []
    let position = 0
    let buffer = ""

    const flushText = () => {
      let content = buffer
      if (content.includes("\r")) content = content.replace(/\r\n?/g, "\n")
      if (content.includes("\u200B")) content = content.replace(/\u200B/g, "")
      buffer = ""
      if (!content) return
      parts.push({ type: "text", content, start: position, end: position + content.length })
      position += content.length
    }

    const pushFile = (file: HTMLElement) => {
      const content = file.textContent ?? ""
      parts.push({
        type: "file",
        path: file.dataset.path!,
        content,
        start: position,
        end: position + content.length,
      })
      position += content.length
    }

    const pushAgent = (agent: HTMLElement) => {
      const content = agent.dataset.content ?? agent.textContent ?? ""
      const id = agent.dataset.id
      parts.push({
        type: "agent",
        name: agent.dataset.name!,
        ...(id ? { id } : {}),
        content,
        start: position,
        end: position + content.length,
      })
      position += content.length
    }

    const pushRule = (rule: HTMLElement) => {
      const content = rule.dataset.content ?? rule.textContent ?? ""
      parts.push({
        type: "rule",
        path: rule.dataset.path!,
        name: rule.dataset.name!,
        ruleType: rule.dataset.ruleType ?? "always_on",
        content,
        start: position,
        end: position + content.length,
      })
      position += content.length
    }

    const visit = (node: Node) => {
      if (node.nodeType === Node.TEXT_NODE) {
        buffer += node.textContent ?? ""
        return
      }
      if (node.nodeType !== Node.ELEMENT_NODE) return

      const el = node as HTMLElement
      if (el.dataset.type === "file") {
        flushText()
        pushFile(el)
        return
      }
      if (el.dataset.type === "agent") {
        flushText()
        pushAgent(el)
        return
      }
      if (el.dataset.type === "rule") {
        flushText()
        pushRule(el)
        return
      }
      if (el.tagName === "BR") {
        buffer += "\n"
        return
      }

      for (const child of Array.from(el.childNodes)) {
        visit(child)
      }
    }

    const children = Array.from(editorRef.childNodes)
    children.forEach((child, index) => {
      const isBlock = child.nodeType === Node.ELEMENT_NODE && ["DIV", "P"].includes((child as HTMLElement).tagName)
      visit(child)
      if (isBlock && index < children.length - 1) {
        buffer += "\n"
      }
    })

    flushText()

    if (parts.length === 0) parts.push(...DEFAULT_PROMPT)
    return parts
  }

  /**
   * 依据当前文本与光标位置刷新 slash 弹窗（开关 + 搜索关键词 + 候选收窄）。
   *
   * 触发判定与候选收窄的完整规则见 ./prompt-input/slash-trigger.ts。
   * 除 input 事件外，光标移动（方向键、Home/End、鼠标点击、选区变化）也要调用它：
   * 移动光标不产生 input 事件，若不同步，关键词会停留在上一次输入时的值 ——
   * 例如 `12 /email` 把光标移到 `/e` 之后，列表仍按 email 过滤而非 e。
   */
  const syncSlashPopover = (input: { rawText: string; cursorPosition: number; parts: Prompt }) => {
    // at 弹层开着时不抢（同 syncSlashOnCaretMove 的守卫）：@ 触发与 slash 触发互斥，
    // 不拦截的话下面的 resetSlashSession 会在同一次 input 事件里把刚打开的 at 弹层关掉。
    if (store.popover === "at") return
    const slashTrigger = resolveSlashTrigger({
      rawText: input.rawText,
      cursor: input.cursorPosition,
    })
    if (!slashTrigger) {
      resetSlashSession()
      return
    }
    // 触发区间若落在已有技能 pill（agent part）的内容范围内，说明 / 来自 pill
    // 而非用户输入（删完 pill 后的文字后 rawText 又变回 /wr），不弹窗。
    const triggerStart = slashTrigger.session.start
    const withinPill = input.parts.some(
      (p) =>
        p.type === "agent" &&
        p.content.startsWith("/") &&
        triggerStart >= p.start &&
        triggerStart < p.end,
    )
    if (withinPill) {
      resetSlashSession()
      return
    }
    // 增量推进而非重设：光标之后可能仍有本会话输入的关键词字符（`/ex｜mail`）。
    const session = advanceSlashSession({
      rawText: input.rawText,
      cursor: input.cursorPosition,
      previous: slashSession(),
      fallback: slashTrigger.session,
    })
    setSlashSession(session)
    // 必须用推进后的 session 判断，不能用 slashTrigger 里的初始区间：光标停在关键词
    // 中间时（`/ex｜mail`），按初始区间会把 mail 误判成其他内容，普通命令被错误过滤。
    setSlashHasOtherContent(hasContentOutsideSession({ rawText: input.rawText, session }))
    slashOnInput(slashTrigger.query)
    // 没有匹配候选时隐藏弹层（不重置会话，用户继续输入匹配时恢复）
    if (slashFlat().length === 0) {
      closePopover()
      return
    }
    setStore("popover", "slash")
  }

  /**
   * 光标移动后刷新检索关键词。
   *
   * 不重新判定触发、也不修改会话区间：移动光标并未改变用户输入过哪些字符，因此替换
   * 范围必须保持 input 事件记录的值。检索关键词则取 `/` 到光标，刻意不受会话末尾
   * 约束 —— 否则同一个可见状态会因输入历史不同而检索结果不同，详见
   * resolveSlashQueryInSession 的说明。
   *
   * 光标退到 `/` 之上或越过空白时只收起弹层、保留区间，移回来要能恢复 —— 因此这里
   * 不能要求弹层当前是开着的。`123 /123` 中光标越过 `/` 再移回来就是这种情况：区间
   * 仍是 [4,8]，无需重建即可安全恢复。用户主动 Escape 时会连区间一起作废，不会被
   * 这里重新唤起。
   */
  const syncSlashOnCaretMove = () => {
    if (store.mode === "shell") return
    if (composing()) return
    // 开着的是别的弹层时不抢；未开弹层（null）则继续，以便从会话内恢复。
    if (store.popover && store.popover !== "slash") return
    const session = slashSession()
    if (!session) return

    const selection = window.getSelection()
    if (!selection || selection.rangeCount === 0 || !editorRef.contains(selection.anchorNode)) return
    // 存在选区时不响应：拖选/Shift+方向键属于选择文本，不是移动插入点。
    if (!selection.isCollapsed) return

    const query = resolveSlashQueryInSession({
      rawText: prompt
        .current()
        .map((p) => ("content" in p ? p.content : ""))
        .join(""),
      cursor: getCursorPosition(editorRef),
      session,
    })
    if (query === undefined) {
      // 只收弹层，保留区间：光标可能只是暂时移出，移回来还要按同一区间继续。
      closePopover()
      return
    }
    slashOnInput(query)
    setStore("popover", "slash")
  }

  /** 仅在纯移动光标的按键后同步；字符输入走 onInput，方向键上下由历史导航处理。 */
  const CARET_MOVE_KEYS = new Set(["ArrowLeft", "ArrowRight", "Home", "End"])
  const handleCaretKeyUp = (event: KeyboardEvent) => {
    if (!CARET_MOVE_KEYS.has(event.key)) return
    // Shift+方向键是扩展选区而非移动插入点，不应触发检索。
    if (event.shiftKey) return
    syncSlashOnCaretMove()
  }

  const handleInput = () => {
    const rawParts = parseFromDOM()
    // 同步苍穹应用下拉挂载点：pill 增删时及时更新
    {
      const triggerEl = editorRef.querySelector<HTMLElement>("[data-cosmic-app-trigger]")
      if (triggerEl) setCosmicAppTriggerMount(triggerEl)
      else if (cosmicAppTriggerMount()) setCosmicAppTriggerMount(null)
    }
    // 检测是否还有带 dev-mode 的技能 pill；没有则清除技能触发的开发模式
    const hasDevModeSkill = rawParts.some((p) => getSkillDevName(p) !== undefined)
    if (!hasDevModeSkill) setSkillDevMode(undefined)
    const images = imageAttachments()
    const cursorPosition = getCursorPosition(editorRef)
    const rawText =
      rawParts.length === 1 && rawParts[0]?.type === "text"
        ? rawParts[0].content
        : rawParts.map((p) => ("content" in p ? p.content : "")).join("")
    const hasNonText = rawParts.some((part) => part.type !== "text")
    const shouldReset = !NON_EMPTY_TEXT.test(rawText) && !hasNonText && images.length === 0

    if (shouldReset) {
      // 输入框已清空，会话必须一并作废，否则残留区间会在下次输入时被错误沿用。
      resetSlashSession()
      resetHistoryNavigation()
      if (prompt.dirty()) {
        mirror.input = true
        prompt.set(DEFAULT_PROMPT, 0)
      }
      queueScroll()
      return
    }

    const shellMode = store.mode === "shell"

    if (!shellMode) {
      // @ 触发只匹配光标所在文本段（anchor 文本节点）：pill 的 content 位于元素节点、
      // 不会拼入该段，天然规避「pill content 拼入 rawText 误匹配」（slash 防护的问题），
      // 也支持已存在 pill 后继续 @ 引用。边界字符类与 build-request-parts 的 mention 解析口径一致。
      const sel = window.getSelection()
      const anchor = sel?.anchorNode
      const atSegment =
        anchor && anchor.nodeType === Node.TEXT_NODE && editorRef.contains(anchor)
          ? (anchor.textContent ?? "").slice(0, sel?.anchorOffset ?? 0)
          : ""
      const atMatch = atSegment.match(/(?:^|[\s([{"'])@(\S*)$/)
      if (atMatch) {
        atOnInput(atMatch[1] ?? "")
        setStore("popover", "at")
      } else if (store.popover === "at") {
        setStore("popover", null)
      }
      syncSlashPopover({ rawText, cursorPosition, parts: rawParts })
    } else {
      resetSlashSession()
    }

    resetHistoryNavigation()

    mirror.input = true
    prompt.set([...rawParts, ...images], cursorPosition)
    queueScroll()
  }

  /**
   * 拦截 contenteditable 的 copy 事件：将 skill/file pill 的 data-content（如
   * /prototype-builder）而非显示名写入剪贴板 text/plain，并将含 data-* 属性的
   * pill DOM 写入 text/html。粘贴时 handlePaste 优先从 text/html 重建 pill，
   * text/plain 作为降级路径。无 pill 时走浏览器默认行为。
   */
  const handleCopy = (event: ClipboardEvent) => {
    const selection = window.getSelection()
    if (!selection || selection.rangeCount === 0) return

    const range = selection.getRangeAt(0)
    if (!editorRef.contains(range.commonAncestorContainer)) return

    const fragment = range.cloneContents()
    if (!fragment.querySelector("[data-type='agent'], [data-type='file']")) return

    let text = ""
    const walk = (node: Node) => {
      if (node.nodeType === Node.TEXT_NODE) {
        text += (node.textContent ?? "").replace(/\u200B/g, "")
        return
      }
      if (node.nodeType !== Node.ELEMENT_NODE) return
      const el = node as HTMLElement
      if (el.dataset.type === "agent" || el.dataset.type === "file" || el.dataset.type === "rule") {
        text += el.dataset.content ?? el.textContent ?? ""
        return
      }
      if (el.tagName === "BR") {
        text += "\n"
        return
      }
      for (const child of Array.from(el.childNodes)) walk(child)
    }

    const children = Array.from(fragment.childNodes)
    children.forEach((child, index) => {
      const isBlock = child.nodeType === Node.ELEMENT_NODE && ["DIV", "P"].includes((child as HTMLElement).tagName)
      walk(child)
      if (isBlock && index < children.length - 1) text += "\n"
    })

    const tempDiv = document.createElement("div")
    tempDiv.appendChild(fragment)
    const html = tempDiv.innerHTML

    event.preventDefault()
    event.clipboardData?.setData("text/plain", text)
    event.clipboardData?.setData("text/html", html)
  }

  const addPart = (part: ContentPart) => {
    if (part.type === "image") return false

    const selection = window.getSelection()
    if (!selection) return false

    if (selection.rangeCount === 0 || !editorRef.contains(selection.anchorNode)) {
      editorRef.focus()
      const cursor = prompt.cursor() ?? promptLength(prompt.current())
      setCursorPosition(editorRef, cursor)
    }

    if (selection.rangeCount === 0) return false
    const range = selection.getRangeAt(0)
    if (!editorRef.contains(range.startContainer)) return false

    if (part.type === "file" || part.type === "agent" || part.type === "rule") {
      const cursorPosition = getCursorPosition(editorRef)
      const rawText = prompt
        .current()
        .map((p) => ("content" in p ? p.content : ""))
        .join("")
      const textBeforeCursor = rawText.substring(0, cursorPosition)
      const query = part.type === "agent" && part.content.startsWith("/")
        ? textBeforeCursor.match(/\/(\S*)$/)
        : textBeforeCursor.match(/@(\S*)$/)
      const pill = createPill(part)
      // 使用零宽空格作为 gap：UI 上不可见，parseFromDOM 会移除它（不保留在 prompt 数据中），
      // renderEditor 在 pill 后追加 \u200B 用于光标定位。sendFollowupDraft 负责在发送时
      // 确保 slash agent pill 后面有空格分隔（后端 ACP 适配层用 split(/\s+/) 解析 command）。
      const gap = document.createTextNode("\u200B")

      if (query) {
        const start = query.index ?? cursorPosition - query[0].length
        setRangeEdge(editorRef, range, "start", start)
        setRangeEdge(editorRef, range, "end", cursorPosition)
      }

      range.deleteContents()
      range.insertNode(gap)
      range.insertNode(pill)
      range.setStartAfter(gap)
      range.collapse(true)
      selection.removeAllRanges()
      selection.addRange(range)
    }

    if (part.type === "text") {
      const fragment = createTextFragment(part.content)
      const last = fragment.lastChild
      range.deleteContents()
      range.insertNode(fragment)
      if (last) {
        if (last.nodeType === Node.TEXT_NODE) {
          const text = last.textContent ?? ""
          if (text === "\u200B") {
            range.setStart(last, 0)
          }
          if (text !== "\u200B") {
            range.setStart(last, text.length)
          }
        }
        if (last.nodeType !== Node.TEXT_NODE) {
          const isBreak = last.nodeType === Node.ELEMENT_NODE && (last as HTMLElement).tagName === "BR"
          const next = last.nextSibling
          const emptyText = next?.nodeType === Node.TEXT_NODE && (next.textContent ?? "") === ""
          if (isBreak && (!next || emptyText)) {
            const placeholder = next && emptyText ? next : document.createTextNode("\u200B")
            if (!next) last.parentNode?.insertBefore(placeholder, null)
            placeholder.textContent = "\u200B"
            range.setStart(placeholder, 0)
          } else {
            range.setStartAfter(last)
          }
        }
      }
      range.collapse(true)
      selection.removeAllRanges()
      selection.addRange(range)
    }

    handleInput()
    // 必须在 handleInput 之后作废：那里会重新判定触发并可能建立新会话，
    // 只关弹层会留下「区间还在、弹层已收」的状态，光标一移动弹层就会自己回来。
    resetSlashSession()
    return true
  }

  const addToHistory = (prompt: Prompt, mode: "normal" | "shell") => {

    const setCurrentHistory = mode === "shell" ? setShellHistory : setHistory

    if (next === currentHistory.entries) return
    setCurrentHistory("entries", next)
  }

  createEffect(
    on(
      () => props.edit?.id,

        if (!id || !edit) return

        for (const item of prompt.context.items()) {
          prompt.context.remove(item.key)
        }

            preview: item.preview,
          })

        setStore("popover", null)
        setStore("historyIndex", -1)
        setStore("savedPrompt", null)
        prompt.set(edit.prompt, promptLength(edit.prompt))

          editorRef.focus()

      },

  }

  const { addAttachments, removeAttachment, handlePaste } = createPromptAttachments({
    editor: () => editorRef,
    isDialogActive: () => !!dialog.active,
    setDraggingType: (type) => setStore("draggingType", type),

      setCursorPosition(editorRef, promptLength(prompt.current()))
    },

    parseTextParts: (text) => {
      if (text.includes("\n") || text.includes("\r")) return [{ type: "text", content: text, start: 0, end: 0 }]
      const match = text.match(/^\/(\S+)\s+([\s\S]+)$/)
      if (!match) return [{ type: "text", content: text, start: 0, end: 0 }]
      const trigger = match[1]
      if (!trigger) return [{ type: "text", content: text, start: 0, end: 0 }]
      const cmd = slashCommands().find((item) => item.trigger === trigger)
      if (!cmd) return [{ type: "text", content: text, start: 0, end: 0 }]
      const body = match[2] ?? ""
      return [
        { type: "agent", name: trigger, content: `/${trigger}`, start: 0, end: 0 },
        { type: "text", content: body ? ` ${body}` : " ", start: 0, end: 0 },
      ]
    },
    readClipboardImage: platform.readClipboardImage,
    getCurrentModelCapabilities: () => {
      const model = local.model.current()
      if (!model?.capabilities) return undefined
      return {

  })

  const modelItems = createMemo(() =>

        local.model.visible({ modelID: item.id, providerID: item.provider.id, lingeeModelId: item.lingeeModelId }),

  const accepting = createMemo(() => {

  })

  const { abort, handleSubmit } = createPromptSubmit({

    },
    setMode: (mode) => setStore("mode", mode),
    setPopover: (popover) => setStore("popover", popover),
    newSessionWorktree: () => {

      const routeDir = routeProjectDir()

      return props.newSessionWorktree
    },
    onNewSessionWorktreeReset: props.onNewSessionWorktreeReset,
    newSessionBaseBranch: () => props.newSessionBaseBranch,
    // SolidJS 的 props 是 getter：直接写 `props.shouldQueue` 只会在组件创建时求值一次，
    // 值随即被冻住。而 session.tsx 只在「已有会话 id 且非子会话」时才传 followup，
    // /session/:id? 又是同一条路由（新任务 → 会话不重新挂载），于是新会话页发出的第一条
    // 消息会把 shouldQueue 永久捕获成 undefined，导致 submit.ts 的入队分支永不成立。
    // 必须用 accessor 包一层，读取时才向上取值。
    shouldQueue: createMemo(() => props.shouldQueue?.() ?? false),
    onQueue: (draft) => props.onQueue?.(draft),
    onAbort: () => props.onAbort?.(),
    onSubmit: props.onSubmit,
    onSent: (sent) => props.onSent?.(sent),
    canUseSkill: (skillName) => canShowSkillTab(licensePlan(), skillName),
    // 技能开发 pill 优先于 starter 选择绑定 partner：否则普通新会话里用 /kingdee-skill-creator
    // 建的会话落默认 kingdee-design，后端 partner 门禁不注入 query_my_tools 等专属工具
    aiPartnerID: () => SKILL_PARTNER_MAP[currentSkillTrigger() ?? ""] ?? selectedStarterAgentID(),

    starterAgentIDFromQuery: () => normalizeStarterAgentID(searchParams.starterAgentID),
    skillInfo: () => {

      return currentSkillTriggers()
        .filter((trigger): trigger is string => isSkillEntryTab(trigger))
        .map((trigger) => ({ skillId: trigger, devMode: skillDevMode() }))
    },
    // 功能开关埋点：仅在发送时上报（见 submit.ts），切换/勾选动作本身不上报
    specMode: modeChip,
    worktreeToggleState: () =>
      // 仅勾选框可见可用时才上报：非 git 项目下 newSessionWorktree 会被目录推导成
      // 非 "main" 值，直接据它上报等于凭空多一条 reuse（见 utils/worktree-track.ts）

  // auto-submit（如 kdapp-confirm 连续点击）会因 fired 仍为 true 而跳过提交。

      fired = true
      props.onAutoSubmitConsumed?.()

  const handleKeyDown = (event: KeyboardEvent) => {
    // mod+u 已由全局 command 系统（file.attach）统一处理，此处不再重复响应，避免文件选择弹窗触发两次

    if (event.key === "Backspace") {
      const selection = window.getSelection()
      if (selection && selection.isCollapsed) {
        const node = selection.anchorNode
        const offset = selection.anchorOffset

        // Manually delete a preceding pill instead of letting the browser
        // remove the contenteditable=false element. On Windows the browser's
        // default Backspace can move focus outside the editor, making it
        // unable to receive further input.
        const getPill = (el: Node | null): HTMLElement | null =>
          el?.nodeType === Node.ELEMENT_NODE &&
          ((el as HTMLElement).dataset.type === "file" ||
            (el as HTMLElement).dataset.type === "agent" ||
            (el as HTMLElement).dataset.type === "rule")
            ? (el as HTMLElement)
            : null

        if (node && node.nodeType === Node.TEXT_NODE) {
          const text = node.textContent ?? ""
          if (/^\u200B*$/.test(text) && offset <= 1) {
            const pill = getPill(node.previousSibling)
            if (pill) {
              event.preventDefault()
              pill.remove()
              handleInput()
              requestAnimationFrame(() => {
                editorRef.focus()
                setCursorPosition(editorRef, prompt.cursor() ?? 0)
              })
              return
            }
          }
          if (/^\u200B+$/.test(text) && offset > 0) {
            const range = document.createRange()
            range.setStart(node, 0)
            range.collapse(true)
            selection.removeAllRanges()
            selection.addRange(range)
          }
          // When deleting the last character of a text node that follows a pill,
          // replace it with \u200B instead of letting the browser delete it.
          // This keeps a non-empty text anchor after the pill — without it,
          // some webviews (e.g. lingee-yzj) cannot place the caret, causing
          // the editor to lose focus and become unfocusable.
          if (offset === 1 && text.length === 1 && !/^\u200B+$/.test(text)) {
            const pill = getPill(node.previousSibling)
            if (pill) {
              event.preventDefault()
              node.textContent = "\u200B"
              const range = document.createRange()
              range.setStart(node, 0)
              range.collapse(true)
              selection.removeAllRanges()
              selection.addRange(range)
              handleInput()
              return
            }
          }
        }

        if (node === editorRef && offset > 0) {
          const prevChild = editorRef.childNodes[offset - 1]
          // 光标在 \u200B gap 文本节点之后时（focusEditorEnd 的常见落点），
          // 往前再看一个 sibling 找 pill，避免逐字符删 \u200B 再触发浏览器插入 <br>。
          const pill =
            prevChild?.nodeType === Node.TEXT_NODE && /^\u200B+$/.test(prevChild.textContent ?? "")
              ? getPill(editorRef.childNodes[offset - 2] ?? null)
              : getPill(prevChild ?? null)
          if (pill) {
            event.preventDefault()
            pill.remove()
            handleInput()
            requestAnimationFrame(() => {
              editorRef.focus()
              setCursorPosition(editorRef, prompt.cursor() ?? 0)
            })
            return
          }
        }
      }
    }

    if (CUSTOMER_SHELL_MODE_ENABLED && event.key === "!" && store.mode === "normal") {
      const cursorPosition = getCursorPosition(editorRef)
      if (cursorPosition === 0) {
        setStore("mode", "shell")
        setStore("popover", null)
        event.preventDefault()
        return
      }
    }

    if (event.key === "Escape") {
      if (store.popover) {
        // slash 要连会话一起作废：Escape 是用户主动放弃本次触发，若只收弹层，
        // 保留的区间会让后续光标移动把它重新唤起，Escape 就形同失效。
        if (store.popover === "slash") resetSlashSession()
        else closePopover()
        event.preventDefault()
        event.stopPropagation()
        return
      }

      if (store.mode === "shell") {
        setStore("mode", "normal")
        event.preventDefault()
        event.stopPropagation()
        return
      }

      // if (working()) {
      //   abort()
      //   event.preventDefault()
      //   event.stopPropagation()
      //   return
      // }

      if (escBlur()) {
        editorRef.blur()
        event.preventDefault()
        event.stopPropagation()
        return
      }
    }

    if (store.mode === "shell") {
      const { collapsed, cursorPosition, textLength } = getCaretState()
      if (event.key === "Backspace" && collapsed && cursorPosition === 0 && textLength === 0) {
        setStore("mode", "normal")
        event.preventDefault()
        return
      }
    }

    // Handle Shift+Enter BEFORE IME check - Shift+Enter is never used for IME input
    // and should always insert a newline regardless of composition state
    if (event.key === "Enter" && event.shiftKey) {
      addPart({ type: "text", content: "\n", start: 0, end: 0 })
      event.preventDefault()
      return
    }

    if (event.key === "Enter" && isImeComposing(event)) {
      return
    }

    const ctrl = event.ctrlKey && !event.metaKey && !event.altKey && !event.shiftKey

    if (store.popover) {
      if (event.key === "Tab") {
        selectPopoverActive()
        event.preventDefault()
        return
      }
      const nav = event.key === "ArrowUp" || event.key === "ArrowDown" || event.key === "Enter"
      const ctrlNav = ctrl && (event.key === "n" || event.key === "p")
      if (nav || ctrlNav) {
        if (store.popover === "at") {
          atOnKeyDown(event)
          event.preventDefault()
          return
        }
        if (store.popover === "slash") {
          // When the slash popover has no matching results, don't intercept
          // Enter/Arrow keys — let the user submit messages and move the
          // caret normally instead of swallowing the keystrokes.
          if (slashFlat().length === 0) return
          slashOnKeyDown(event)
        }
        event.preventDefault()
        return
      }
    }

    if (ctrl && event.code === "KeyG") {
      if (store.popover) {
        // 与 Escape 同为主动取消，slash 需连会话一起作废，否则光标移动会把弹层唤回。
        if (store.popover === "slash") resetSlashSession()
        else closePopover()
        event.preventDefault()
        return
      }
      if (working()) {
        abort()
        event.preventDefault()
      }
      return
    }

    if (event.key === "ArrowUp" || event.key === "ArrowDown") {
      if (event.altKey || event.ctrlKey || event.metaKey) return
      const { collapsed } = getCaretState()
      if (!collapsed) return

      const cursorPosition = getCursorPosition(editorRef)
      const textContent = prompt
        .current()
        .map((part) => ("content" in part ? part.content : ""))
        .join("")
      const direction = event.key === "ArrowUp" ? "up" : "down"
      if (!canNavigateHistoryAtCursor(direction, textContent, cursorPosition, store.historyIndex >= 0)) return
      if (navigateHistory(direction)) {
        event.preventDefault()
      }
      return
    }

    // Note: Shift+Enter is handled earlier, before IME check
    if (event.key === "Enter" && !event.shiftKey) {
      event.preventDefault()
      if (event.repeat) return
      if (
        working() &&
        prompt
          .current()
          .map((part) => ("content" in part ? part.content : ""))
          .join("")
          .trim().length === 0 &&
        imageAttachments().length === 0 &&
        commentCount() === 0

        return
      }
      handleSubmit(event)
    }

                if (props.newSessionWorktree === "create") {

                  if (!ok) return
                  props.onNewSessionBaseBranchChange?.(name)
                  // 若之前是因为选中占用分支而被自动锁定("main" 以外的目录值),这里回落为
                  // "main",把 checkbox 交还给用户手动控制。
                  if (props.newSessionWorktree !== "create") props.onNewSessionWorktreeChange?.("main")

                    navigateToNewSessionDirectory(effectiveProjectDir())
                    return
                  }
                  void files.tree.refreshLoaded()
                })
              }}
            />
