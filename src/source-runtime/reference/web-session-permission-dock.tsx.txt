import { For, Show } from "solid-js"
import type { PermissionRequest } from "@opencode-ai/sdk/v2"
import { Button } from "@kcode/ui/button"
import { DockPrompt } from "@kcode/ui/dock-prompt"
import { Icon } from "@kcode/ui/icon"
import { useLanguage } from "@/context/language"

export function SessionPermissionDock(props: {
  request: PermissionRequest
  responding: boolean
  onDecide: (response: "once" | "always" | "reject") => void
}) {
  const language = useLanguage()
  const isDeletePermission = () => props.request.permission === "file_delete"
  const metadata = () => props.request.metadata as {
    files?: Array<{ type?: string }>
    sourcePaths?: string[]
    destinationPath?: string
    destinationPaths?: string[]
  }
  const translate = (key: string, vars: Record<string, string | number | boolean>, fallback: string) => {
    const value = language.t(key as Parameters<typeof language.t>[0], vars)
    return value === key ? fallback : value
  }

  const isDanger = () =>
    typeof props.request.metadata?.command === "string" && props.request.metadata.command.length > 0

  const toolDescription = () => {
    const key = `settings.permissions.tool.${props.request.permission}.description`
    const value = language.t(key as Parameters<typeof language.t>[0])
    if (value === key) return ""
    return value
  }

  const requestSummary = () => {
    const data = metadata()
    const deleted = data.files?.filter((item) => item.type === "delete").length ?? 0
    if (deleted > 0) {
      return translate("notification.permission.summary.delete", { count: deleted }, `Deletes ${deleted} file(s)`)
    }
    if (data.sourcePaths?.length && data.destinationPath) {
      return translate(
        "notification.permission.summary.transfer",
        {
          count: data.sourcePaths.length,
          destination: data.destinationPath,
        },
        `Reads ${data.sourcePaths.length} source file(s) and writes to ${data.destinationPath}`,
      )
    }
    if (data.destinationPaths?.length) {
      return translate(
        "notification.permission.summary.destination",
        {
          count: data.destinationPaths.length,
        },
        `Writes to ${data.destinationPaths.length} destination path(s)`,
      )
    }
    return ""
  }

  return (
    <DockPrompt
      kind="permission"
      header={
        <div data-slot="permission-row" data-variant="header">
          <span data-slot="permission-icon">
            <Icon name="warning-triangle" size="normal" />
          </span>
          <div data-slot="permission-header-title" title={isDanger() ? language.t("notification.permission.dangerTitle") : language.t("notification.permission.title")}>
            {isDanger() ? language.t("notification.permission.dangerTitle") : language.t("notification.permission.title")}
          </div>
        </div>
      }
      footer={
        <>
          <div />
          <div data-slot="permission-footer-actions">
            <Button variant="ghost" size="normal" onClick={() => props.onDecide("reject")} disabled={props.responding} title={language.t("ui.permission.deny")}>
              {language.t("ui.permission.deny")}
            </Button>
            <Show when={!isDanger()}>
              <Button
                variant="secondary"
                size="normal"
                onClick={() => props.onDecide("always")}
                disabled={props.responding}
                title={language.t("ui.permission.allowAlways")}
              >
                {language.t("ui.permission.allowAlways")}
              </Button>
            </Show>
            <Button variant="primary" size="normal" onClick={() => props.onDecide("once")} disabled={props.responding} title={language.t("ui.permission.allowOnce")}>
              {language.t("ui.permission.allowOnce")}
            </Button>
          </div>
        </>
      }
    >
      <Show when={!isDanger() && toolDescription()}>
        <div data-slot="permission-row">
          <span data-slot="permission-spacer" aria-hidden="true" />
          <div data-slot="permission-hint" title={toolDescription()}>{toolDescription()}</div>
        </div>
      </Show>

      <Show when={typeof props.request.metadata?.command === "string" && props.request.metadata.command.length > 0}>
        <div data-slot="permission-row">
          <span data-slot="permission-spacer" aria-hidden="true" />
          <code
            class="text-[12px] leading-[20px] text-[var(--neutral-700)]"
            title={props.request.metadata.command as string}
          >
            {props.request.metadata.command as string}
          </code>
        </div>
      </Show>

      <Show when={!isDanger() && props.request.patterns.length > 0}>
        <div data-slot="permission-row">
          <span data-slot="permission-spacer" aria-hidden="true" />
          <div data-slot="permission-patterns">
            <For each={props.request.patterns}>
              {(pattern) => <code class="text-[12px] leading-[20px] text-[var(--neutral-700)]" title={pattern}>{pattern}</code>}
            </For>
          </div>
        </div>
      </Show>
    </DockPrompt>
  )
}
