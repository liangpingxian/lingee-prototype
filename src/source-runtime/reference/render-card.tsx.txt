function isThenable(v: unknown): v is PromiseLike<unknown> {
  return v != null && typeof (v as { then?: unknown }).then === "function"
}

function CardIcon(props: { kind: string }) {
  const icon = () => KIND_ICON_MAP[props.kind] ?? renderCardIcons.general
  return <img src={icon()} alt="" draggable={false} />
}

export function RenderCard(props: {
  kind: string
  config: Record<string, string>
  localTestDisabled?: boolean
  disabled?: boolean
  onCardClick?: (kind: string, config: Record<string, string>) => unknown
}) {
  // Phase-only kdapp blocks only trigger the sidebar drawer, no chat card
  if (props.kind === "kdapp" && props.config.phase && !props.config.content && !props.config.url) {
    return null
  }

  // 宿主为该 kind 提供了渲染组件时交给宿主（技能恢复卡的「重新准备」要经宿主的发送链路发出消息）。
  const slot = useRenderCardSlots()[props.kind]
  if (slot) return <Dynamic component={slot} kind={props.kind} config={props.config} />
  // 没有宿主处理器时不出恢复卡，保留工具原有的文字输出。
  if (props.kind === SKILL_RECOVERY_KIND) return null

  const i18n = useI18n()
  const desc = () => props.config.desc || props.config.description || i18n.t(DESC_KEYS[props.kind] ?? DESC_KEYS.kdform!)
  const url = () => props.config.url ?? ""

  const isPreview = props.kind === "kdapp-artifact" && props.config.phase === "preview"

  // 通用应用产物卡片：仅前端产物会带 url。预览按钮与卡片主体点击（定位工程）是两个都有效的
  // 动作，因此不复用 kdapp-artifact 的 isPreview gate —— 后者会把右侧箭头隐藏掉。
  const hasStandalonePreview = () => props.kind === "kdstandalone" && !!props.config.url

  const [previewPending, setPreviewPending] = createSignal(false)
  const PREVIEW_LOCK_FALLBACK_MS = 5000
  const triggerPreview = () => {
    if (previewPending()) return
    const jid = props.config.jobId ?? props.config.serviceId
    if (!jid) return
    setPreviewPending(true)
    const release = () => setPreviewPending(false)
    const result = props.onCardClick?.("kdapp-preview-start", { jobId: jid, url: props.config.previewUrl ?? props.config.url ?? "", appCode: props.config.appCode ?? "" })
    if (isThenable(result)) {
      result.then(release, release)
    } else {
      setTimeout(release, PREVIEW_LOCK_FALLBACK_MS)
    }
  }

  const open = () => {
    console.info("[render-card] open clicked", {
      kind: props.kind,
      hasUrl: !!props.config.url,
      hasMetadata: !!props.config.metadata,
      hasPlugins: !!props.config.plugins,
      hasApi: !!props.config.api,
      disabled: props.disabled,
      localTestDisabled: props.localTestDisabled,
      isPreview,
      hasOnCardClick: !!props.onCardClick,
    })
    if (props.disabled) return
    if (isPreview) {
      console.info("[render-card] → triggerPreview")
      triggerPreview()
      return
    }
    if (props.onCardClick) {
      console.info("[render-card] → onCardClick", { kind: props.kind })
      props.onCardClick(props.kind, props.config)
      return
    }
    const href = url()
    console.info("[render-card] → fallback, trying bridge", { href })
    if (href) {
      const lingeeBridge = (window as { lingeeBridge?: { previewBrowser?: { openUrl?: (input: { url: string }) => Promise<unknown>; openSurface?: (input: { tabId: string; url: string; partition: string }) => Promise<unknown> } } }).lingeeBridge
      const previewBrowser = lingeeBridge?.previewBrowser
      if (previewBrowser?.openUrl) {
        // 生产环境 (lingee): openUrl → preview-browser:open 事件 → PreviewBrowserPanel
        void previewBrowser.openUrl({ url: href })
      } else if (previewBrowser?.openSurface) {
        // 开发环境 (kwork-desktop): openSurface → IPC → PreviewBrowserDock
        void previewBrowser.openSurface({ tabId: 'preview-card', url: href, partition: 'persist:lingee-preview-browser-card' })
      } else {
        window.open(href, "_blank", "noopener,noreferrer")
      }
    }
  }

  // 专家阶段卡：展示角色头像、阶段标题与运行状态，不走产物预览卡的点击逻辑。
  if (props.kind === "kdexpert") {
    const role = () => props.config.role ?? ""
    const rawStatus = () => props.config.status ?? "start"
    const sectionStatus = () => rawStatus() === "complete"
      ? "completed"
      : rawStatus() === "failed" || rawStatus() === "error"
        ? "error"
        : "running"
    const statusLabel = () => rawStatus() === "complete"
      ? i18n.t("ui.tool.task.status.completed")
      : rawStatus() === "failed" || rawStatus() === "error"
        ? i18n.t("ui.tool.task.status.error")
        : i18n.t("ui.tool.task.status.running")
    const isActive = () => rawStatus() === "start" || rawStatus() === "running"
    const avatar = () => pickAvatar(role() || props.config.title || "expert")

    return (
      <div data-component="expert-card">
        <div data-component="expert-card-header">
          <span data-component="expert-card-avatar" data-tool-status={sectionStatus()} aria-hidden="true">
            <img src={avatar()} alt="" data-component="expert-card-avatar-image" draggable={false} />
          </span>
          <span data-component="expert-card-role" title={role()}>{role()}</span>
          <span data-component="expert-card-status" data-tool-status={sectionStatus()}>
            <TextShimmer text={statusLabel()} active={isActive()} />
          </span>
        </div>
        <Show when={props.config.title}>
          <div data-component="expert-card-title" title={props.config.title}>{props.config.title}</div>
        </Show>
      </div>
    )
  }

  return (
    <div
      data-component="render-card"
      data-preview={isPreview ? "" : undefined}
      data-disabled={props.disabled ? "" : undefined}
      aria-disabled={props.disabled ? "true" : undefined}
      onClick={open}
    >
      <div data-slot="render-card-icon">
        <CardIcon kind={props.kind} />
      </div>
      <div data-slot="render-card-body">
        <span data-slot="render-card-title">{props.config.title ?? props.config.displayName ?? ""}</span>
        <span data-slot="render-card-desc">{desc()}</span>
      </div>
      <Show when={props.kind === "kdapp"}>
        <span
          data-slot="render-card-confirm"
          onClick={(e) => {
            e.stopPropagation()
            props.onCardClick?.("kdapp-confirm", props.config)
          }}
        >
          {i18n.t("ui.renderCard.confirm")}
        </span>
      </Show>
      <Show when={isPreview}>
        <span
          data-slot="render-card-preview-toggle"
          data-pending={previewPending() ? "" : undefined}
          onClick={(e) => {
            e.stopPropagation()
            triggerPreview()
          }}
        >
          {i18n.t("ui.renderCard.previewStart")}
        </span>
        <span
          data-slot="render-card-detail"
          onClick={(e) => {
            e.stopPropagation()
            props.onCardClick?.("kdapp-artifact", props.config)
          }}
        >
          {i18n.t("ui.renderCard.detail")}
        </span>
      </Show>
      <Show when={hasStandalonePreview()}>
        <span
          data-slot="render-card-preview-toggle"
          onClick={(e) => {
            e.stopPropagation()
            props.onCardClick?.("kdstandalone-preview", props.config)
          }}
        >
          {i18n.t("ui.renderCard.previewStart")}
        </span>
      </Show>
      <Show when={props.kind === "ai-partner" || props.kind === "kdskill"}>
        <span
          data-slot="render-card-local-test"
          data-disabled={props.localTestDisabled ? "" : undefined}
          aria-disabled={props.localTestDisabled ? "true" : undefined}
          onClick={(e) => {
            e.preventDefault()
            e.stopPropagation()
            if (props.localTestDisabled) return
            props.onCardClick?.(`${props.kind}-local-test`, props.config)
          }}
        >
          <Icon name="monitor" size="small" />
          {i18n.t("ui.renderCard.localTest")}
        </span>
      </Show>
      <Show when={!isPreview}>
        <div data-slot="render-card-arrow">
          <ArrowIcon />
        </div>
      </Show>
    </div>
  )
}
